# here is the list of instructions it supports currently (not tested)

0x0: nop, hlt, sez, clz, sec, clc, cmc, clf

0x1: psh, pop, sic, sdc, trs, tsr, php, plp, rsp, csp, clr

0x2: add, sub, inc, dec

0x3: cmp, cmn, tst, teq

0x4: ior, nor, xor, xnr, and, nad

0x5: lsl, lsr, rol, ror, sar, flp, pct

0x6: mul, iml, muh, imh

0x7: div, idv, mod, imd

0x8: ldi

0x9: mov, mar

0xA: ldr, str

0xB: rlj

0xC: jmp

0xD: cal, ret
